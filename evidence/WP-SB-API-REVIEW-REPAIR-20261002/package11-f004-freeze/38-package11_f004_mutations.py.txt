def native(label, cwd, command, expected=0):
    phase = 'package11-f004-mut-' + label
    completed = subprocess.run([sys.executable, str(RUN), phase, str(cwd), '--', *command],
                               stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    receipt = next(json.loads(line) for line in reversed((E / 'VERIFICATION_LOG.jsonl').read_text().splitlines())
                   if json.loads(line)['phase'] == phase)
    assert sha(receipt['log']) == receipt['log_sha256']
    print(phase, receipt['exit_code'], flush=True)
    if completed.returncode != expected or receipt['exit_code'] != expected:
        raise RuntimeError(completed.stdout[-7000:])
    return receipt


def build(label, sources):
    for key, rel in PATHS.items():
        (M / rel).write_text(sources[key])
    receipt = native(label + '-build', M,
        ['dotnet', 'build', 'src/ViciOne.ServiceBus/ViciOne.ServiceBus.csproj', '-c', 'Release', '--no-restore', '-warnaserror'])
    folder = R / 'repair-evidence/package11-f004-binaries' / label
    folder.mkdir(parents=True, exist_ok=False)
    source_hashes = {}
    for key, rel in PATHS.items():
        snapshot = folder / (key + '.cs.txt')
        snapshot.write_bytes((M / rel).read_bytes())
        source_hashes[rel] = dict(path=str(snapshot), sha256=sha(snapshot))
    binaries = []
    for dll in DLLS:
        compiled = M / 'artifacts/sdk/bin' / dll.removesuffix('.dll') / 'release' / dll
        snapshot = folder / dll
        shutil.copyfile(compiled, snapshot)
        shutil.copyfile(compiled, OUT / dll)
        binaries.append(dict(path=str(snapshot), sha256=sha(snapshot)))
    return dict(receipt=receipt, binaries=binaries, source_snapshots=source_hashes)

def replace_once(sources, key, before, after):
    assert sources[key].count(before) == 1, (key, before)
    sources[key] = sources[key].replace(before, after)


def mutation(name):
    sources = dict(corrected)
    negative, positive = 'custom-factory-scope', 'nondi-outgoing'
    if name == 'process-meter-selected':
        replace_once(sources, 'journal_telemetry', 'meterFactory!.Create(new MeterOptions(ServiceBusTelemetry.MeterName) { Version = version })', 'new Meter(ServiceBusTelemetry.MeterName, version)')
    elif name == 'factory-failure-global-fallback':
        replace_once(sources, 'journal_telemetry', '            try { processMeter?.Dispose(); } catch { }', '            try { processMeter?.Dispose(); } catch { }\n            if (!processOwned)\n            {\n                _operations = Process._operations;\n                _duration = Process._duration;\n                _activitySource = Process._activitySource;\n            }')
        negative, positive = 'factory-failure-after-success', 'custom-factory-scope'
    elif name == 'borrowed-factory-meter-disposed':
        replace_once(sources, 'journal_telemetry', '            _processMeter = processMeter;', '            _processMeter = meter;')
        negative, positive = 'borrowed-factory-not-disposed', 'custom-factory-scope'
    elif name == 'owned-source-disposal-omitted':
        replace_once(sources, 'journal_telemetry', '        try { _activitySource?.Dispose(); } catch { }', '')
        negative, positive = 'activity-source-disposal', 'custom-factory-scope'
    elif name == 'registration-caches-first-telemetry':
        replace_once(sources, 'journal_registration', '    public Type BusType => typeof(TBus);', '    MessageJournalTelemetry? _cachedTelemetry;\n\n    public Type BusType => typeof(TBus);')
        replace_once(sources, 'journal_registration', '            provider.GetRequiredService<MessageJournalTelemetry>());', '            _cachedTelemetry ??= provider.GetRequiredService<MessageJournalTelemetry>());')
        negative, positive = 'two-roots-same-service-collection', 'two-independent-providers'
    elif name == 'di-writer-forgets-owner':
        replace_once(sources, 'journal_registration', '        var writer = new MessageJournalWriter(store, policy, options,\n            provider.GetRequiredService<MessageJournalTelemetry>());', '        var writer = new MessageJournalWriter(store, policy, options);')
        positive = 'nondi-consume'
    elif name == 'shared-transport-graph-restored':
        sources['inmemory_factory'] = original['inmemory_factory'].decode()
        negative, positive = 'two-roots-same-service-collection', 'two-independent-providers'
    else:
        raise ValueError(name)
    return sources, negative, positive

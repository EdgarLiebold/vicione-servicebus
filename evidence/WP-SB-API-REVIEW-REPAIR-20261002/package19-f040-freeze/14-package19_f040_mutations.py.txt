W = Path('/Users/edgar.liebold/Downloads/ViciOne Suite 2.0')

R = Path('/private/tmp/vicione-servicebus-api-review-20261001')
M = R / 'mutations/source-main'
F = R / 'frozen-repository'
E = O / 'evidence/WP-SB-API-REVIEW-REPAIR-20261002'
RUN = W / 'SERVICEBUS_API_REVIEW_AND_REPAIR/run_operation.py'
C = R / 'repair-consumers/package19-transport/corrected'
OUT = C / 'bin/Release/net10.0'
APP = 'ActivityOwnership.PublicConsumer.dll'
DLL = 'ViciOne.ServiceBus.dll'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
PATHS = json.loads((E / 'PACKAGE18_F035_MUTATIONS.json').read_text())['frame']
PATHS.update(inmemory_receive_context='src/ViciOne.ServiceBus/InMemoryTransport/Runtime/InMemoryReceiveEndpointContext.cs')
original = {key: (F / rel).read_bytes() if (F / rel).exists() else None for key, rel in PATHS.items()}
assert all(((M / rel).read_bytes() if (M / rel).exists() else None) == original[key] for key, rel in PATHS.items()), 'Mutation frame must start frozen original'
corrected = {key: (O / rel).read_text() for key, rel in PATHS.items()}

runtime_sha256 = hashlib.sha256(runtime).hexdigest()
oracle = {str(p): sha(p) for p in C.iterdir() if p.is_file()}
oracle[str(OUT / APP)] = sha(OUT / APP)
constants = {str(p): sha(p) for p in OUT.glob('*.dll') if p.name != DLL}
result = dict(status='RUNNING', oracle_hashes=oracle, constant_dependencies=constants, frame=PATHS, selected_runtime_sha256=runtime_sha256, controls=[], mutants=[])

def replace_once(text, before, after):
    assert text.count(before) == 1, (before, text.count(before))
    return text.replace(before, after, 1)

def mutation(name):
    texts = dict(corrected)
    target = 'inmemory_receive_context'
    before = '        _configuration.InputAddress.ParseHostPathAndEntityName(out _, out var name);'
    if name == 'RestoreEscapedTopologyName':
        after = '        var name = _configuration.InputAddress.GetEndpointName();'
    elif name == 'DecodeEntityTwice':
        after = before + '\n        name = Uri.UnescapeDataString(name);'
    elif name == 'UseSenderWildcardParser':
        after = '        var name = new InMemoryEndpointAddress(_hostConfiguration.HostAddress, _configuration.InputAddress).Name;'
    else:
        raise ValueError(name)
    texts[target] = replace_once(texts[target], before, after)
    assert sum(texts[k] != corrected[k] for k in texts) == 1
    return texts

def install(texts):
    for key, rel in PATHS.items():
        path = M / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        payload = texts[key].encode()
        if not path.exists() or path.read_bytes() != payload:
            path.write_bytes(payload)

def operation(suffix, cwd, args, expected=0):
    phase = 'package19-f040-mut-' + suffix
    completed = subprocess.run(['python3', str(RUN), phase, str(cwd), '--', *args], check=False)
    assert completed.returncode == expected, (phase, completed.returncode, expected)
    receipt = next(json.loads(line) for line in reversed((E / 'VERIFICATION_LOG.jsonl').read_text().splitlines()) if json.loads(line)['phase'] == phase)
    assert receipt['exit_code'] == expected and sha(receipt['log']) == receipt['log_sha256']
    text = Path(receipt['log']).read_text()
    if 'build' in args:
        assert '0 Warnung(en)' in text and '0 Fehler' in text
    return receipt

def build(name, texts):
    install(texts)
    receipt = operation(name + '-build', M, ['dotnet', 'build', 'src/ViciOne.ServiceBus/ViciOne.ServiceBus.csproj', '-c', 'Release', '--no-restore', '-warnaserror'])
    directory = R / 'repair-evidence/package19-f040-binaries' / name
    directory.mkdir(parents=True, exist_ok=False)
    snapshots = []
    for index, (key, rel) in enumerate(PATHS.items()):
        snapshot = directory / ('source-' + str(index) + '.txt')
        snapshot.write_bytes((M / rel).read_bytes())
        snapshots.append(dict(key=key, relative=rel, path=str(snapshot), sha256=sha(snapshot), current_source_sha256=hashlib.sha256(texts[key].encode()).hexdigest()))
    binary = M / 'artifacts/sdk/bin/ViciOne.ServiceBus/release' / DLL
    shutil.copy2(binary, directory / DLL)
    # Product Release images carry their portable PDB embedded in the PE.
    shutil.copy2(binary, OUT / DLL)
    return dict(name=name, build=receipt, binary=dict(path=str(directory / DLL), sha256=sha(directory / DLL)), source_snapshots=snapshots)

def case(suffix, mode, expected=0, assertion=None, assertion_line=None):
    receipt = operation(suffix, C, ['dotnet', 'bin/Release/net10.0/' + APP, mode, '00:00:12'], expected)
    text = Path(receipt['log']).read_text()
    if mode == 'all':
        assert text.count('\nPASS ') == 14 and 'TOTAL 14 PASS 0 FAIL' in text and '\nFAIL ' not in text
    elif expected == 0:
        assert '\nPASS ' + mode in text and '\nFAIL ' not in text
    else:
        assert '\nFAIL ' + mode in text and assertion in text and 'TimeoutException' not in text and 'Case and cleanup failed' not in text
        assert assertion_line is not None and 'Cases.cs:line ' + str(assertion_line) in text, 'Require exact finite topology assertion frame'
        receipt['finite_topology_assertion_line'] = assertion_line
    for path, digest in {**oracle, **constants}.items():
        assert sha(path) == digest, path
    return receipt

variants = [
    ('RestoreEscapedTopologyName', 'default-backtick', 'default-ascii', 'Xunit.Sdk.SingleException', 79),
    ('DecodeEntityTwice', 'typed-percent', 'typed-ascii', 'Xunit.Sdk.SingleException', 83),
    ('UseSenderWildcardParser', 'default-wildcard', 'default-ascii', 'Xunit.Sdk.SingleException', 83)
]

try:
    stage = build('corrected', corrected)
    stage['all_cases'] = case('corrected-fourteen', 'all')
    result['controls'].append(stage)
    for name, negative, positive, assertion, assertion_line in variants:
        stage = build(name, mutation(name))

    stage['all_cases'] = case('rollback-fourteen', 'all')

    for key, rel in PATHS.items():

    assert all(sha(path) == digest for path, digest in {**oracle, **constants}.items())
    rows = list(csv.DictReader((R / 'FILE_COVERAGE.csv').open()))
    mismatches = [row['path'] for row in rows if sha(M / row['path']) != row['sha256']]
    assert not mismatches, mismatches
    result['mutation_tree_original_files_verified'] = len(rows)
    result['oracle_and_runtime_restored'] = True
    (E / 'PACKAGE19_F040_MUTATIONS.json').write_text(json.dumps(result, indent=2) + '\n')

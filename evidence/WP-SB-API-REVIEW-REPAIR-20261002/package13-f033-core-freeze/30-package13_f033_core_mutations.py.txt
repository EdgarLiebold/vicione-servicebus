def native(label, cwd, command, expected=0):
    phase = 'package13-f033-core-mut-' + label
    completed = subprocess.run([sys.executable, str(RUN), phase, str(cwd), '--', *command],
                               stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    receipt = next(json.loads(line) for line in reversed((E / 'VERIFICATION_LOG.jsonl').read_text().splitlines())
                   if json.loads(line)['phase'] == phase)
    assert sha(receipt['log']) == receipt['log_sha256']
    print(phase, receipt['exit_code'], flush=True)
    if completed.returncode != expected or receipt['exit_code'] != expected:
        raise RuntimeError(completed.stdout[-7000:])
    return receipt


def build(label, sources):
    for key, rel in PATHS.items():
        (M / rel).write_text(sources[key])
    receipt = native(label + '-build', M,
        ['dotnet', 'build', 'src/ViciOne.ServiceBus/ViciOne.ServiceBus.csproj', '-c', 'Release', '--no-restore', '-warnaserror'])
    folder = R / 'repair-evidence/package13-f033-core-binaries' / label
    folder.mkdir(parents=True, exist_ok=False)
    source_hashes = {}
    for key, rel in PATHS.items():
        snapshot = folder / (key + '.cs.txt')
        snapshot.write_bytes((M / rel).read_bytes())
        source_hashes[rel] = dict(path=str(snapshot), sha256=sha(snapshot))
    binaries = []
    for dll in DLLS:
        compiled = M / 'artifacts/sdk/bin' / dll.removesuffix('.dll') / 'release' / dll
        snapshot = folder / dll
        shutil.copyfile(compiled, snapshot)
        shutil.copyfile(compiled, OUT / dll)
        binaries.append(dict(path=str(snapshot), sha256=sha(snapshot)))
    return dict(receipt=receipt, binaries=binaries, source_snapshots=source_hashes)

def mutation(name):
    sources = dict(corrected)
    factory = sources['scope_factory']
    lifetime = sources['owned_lifetime']
    if name == 'original-scope-factory':
        sources['scope_factory'] = original['scope_factory'].decode()
        return sources, ['dual-default'], ['cleanup-only-default']
    if name == 'primary-capture-dropped':
        before = '            operationFailure = exception;'
        assert factory.count(before) == 1
        sources['scope_factory'] = factory.replace(before, '            operationFailure = null;\n            _ = exception;')
        return sources, ['dual-typed'], ['cleanup-only-default']
    if name == 'dual-failure-order-reversed':
        before = '                operationFailure,\n                releaseFailure);'
        assert lifetime.count(before) == 1
        sources['owned_lifetime'] = lifetime.replace(before, '                releaseFailure,\n                operationFailure);')
        return sources, ['dual-default'], ['operation-only-default']
    if name == 'cleanup-failure-swallowed':
        before = '            releaseFailure = exception;'
        assert lifetime.count(before) == 1
        sources['owned_lifetime'] = lifetime.replace(before, '            _ = exception;')
        return sources, ['cleanup-only-default'], ['operation-only-default']
    if name == 'container-consumer-disposed-twice':
        before = '        await OwnedConsumerLifetime.ReleaseAfterOperationAsync(scope, operationFailure).ConfigureAwait(false);'
        assert factory.count(before) == 1
        added = """        if (operationFailure is not null && scope.Context.Consumer is IDisposable disposable)
        {
            try { disposable.Dispose(); } catch { }
        }
"""
        sources['scope_factory'] = factory.replace(before, added + before)
        return sources, ['dual-default'], ['success-default']
    if name == 'mixed-cancellation-collapsed':
        before = '        await OwnedConsumerLifetime.ReleaseAfterOperationAsync(scope, operationFailure).ConfigureAwait(false);'
        assert factory.count(before) == 1
        after = """        try
        {
            await OwnedConsumerLifetime.ReleaseAfterOperationAsync(scope, operationFailure).ConfigureAwait(false);
        }
        catch (AggregateException) when (operationFailure is OperationCanceledException)
        {
            System.Runtime.ExceptionServices.ExceptionDispatchInfo.Capture(operationFailure).Throw();
        }"""
        sources['scope_factory'] = factory.replace(before, after)
        return sources, ['dependency-cancel-dual-default'], ['dependency-cancel-only-default']
    raise ValueError(name)


try:

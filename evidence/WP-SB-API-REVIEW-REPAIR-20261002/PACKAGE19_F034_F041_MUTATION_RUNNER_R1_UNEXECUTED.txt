def replace_once(text, before, after):
    assert text.count(before) == 1, (before, text.count(before))
    return text.replace(before, after, 1)

def mutation(name):
    texts = dict(corrected)
    if name in ['DropPlannedCompanionName', 'IgnoreCapturedCompanionName']:
        if name == 'DropPlannedCompanionName':
            target = 'endpoint_context'
            before = '(IBusRegistrationContext)RegistrationContext, endpointName);'
            after = '(IBusRegistrationContext)RegistrationContext);'
        else:
            target = 'context_definition'
            before = 'return _plannedEndpointName ?? _endpointDefinition.GetEndpointName(formatter);'
            after = 'return _endpointDefinition.GetEndpointName(formatter);'
        texts[target] = replace_once(texts[target], before, after)
    else:
        full = 'Full' in name or name == 'BypassTypedSelector'
        compensate = 'Compensate' in name
        target = 'activity_registration' if full else 'execute_registration'
        kind, argument, variable = ('Compensate', 'TLog', 'compensate') if compensate else ('Execute', 'TArguments', 'execute')
        concrete = kind + 'ActivityEndpointDefinition<TActivity, ' + argument + '>'
        generic = 'I' + kind + 'Activity<' + argument + '>'
        call = '_selector.GetDefinition<' + concrete + '>(provider)'
        if name.startswith('Restore'):
            texts[target] = replace_once(texts[target], call, '_selector.GetEndpointDefinition<' + generic + '>(provider)')
        elif name == 'BypassTypedSelector':
            texts[target] = replace_once(texts[target], call, 'provider.GetService(typeof(' + concrete + ')) as ' + concrete)
        elif name.startswith('AcceptForeign'):
            condition = '&& endpointType != typeof(' + concrete + ')'
            texts[target] = replace_once(texts[target], condition, '&& false')
        elif name.startswith('RejectOwn'):
            condition = '\n                        && endpointType != typeof(' + concrete + ')'
            texts[target] = replace_once(texts[target], condition, '')
        elif name.startswith('DropGenuine'):
            marker = 'Type endpointType = ' + variable + 'EndpointDefinition.GetType();'
            additional = '\n                    if (!endpointType.IsConstructedGenericType\n                        || endpointType.GetGenericTypeDefinition() != typeof(' + kind + 'ActivityEndpointDefinition<,>))\n                        ' + variable + 'EndpointDefinition = null;'
            texts[target] = replace_once(texts[target], marker, marker + additional)
        else:
            raise ValueError(name)
    assert sum(texts[k] != corrected[k] for k in texts) == 1
    return texts

def install(texts):
    for key, rel in PATHS.items():
        path = M / rel
        path.parent.mkdir(parents=True, exist_ok=True)
        payload = texts[key].encode()
        if not path.exists() or path.read_bytes() != payload:
            path.write_bytes(payload)

def operation(suffix, cwd, args, expected=0):
    phase = 'package19-f034-f041-mut-' + suffix
    completed = subprocess.run(['python3', str(RUN), phase, str(cwd), '--', *args], check=False)
    assert completed.returncode == expected, (phase, completed.returncode, expected)
    receipt = next(json.loads(line) for line in reversed((E / 'VERIFICATION_LOG.jsonl').read_text().splitlines()) if json.loads(line)['phase'] == phase)
    assert receipt['exit_code'] == expected and sha(receipt['log']) == receipt['log_sha256']
    text = Path(receipt['log']).read_text()
    if 'build' in args:
        assert '0 Warnung(en)' in text and '0 Fehler' in text
    return receipt

def build(name, texts):
    install(texts)
    receipt = operation(name + '-build', M, ['dotnet', 'build', 'src/ViciOne.ServiceBus.Courier/ViciOne.ServiceBus.Courier.csproj', '-c', 'Release', '--no-restore', '-warnaserror'])
    directory = R / 'repair-evidence/package19-f034-f041-binaries' / name
    directory.mkdir(parents=True, exist_ok=False)
    snapshots = []
    for index, (key, rel) in enumerate(PATHS.items()):
        snapshot = directory / ('source-' + str(index) + '.txt')
        snapshot.write_bytes((M / rel).read_bytes())
        snapshots.append(dict(key=key, relative=rel, path=str(snapshot), sha256=sha(snapshot), current_source_sha256=hashlib.sha256(texts[key].encode()).hexdigest()))
    binaries = []
    for dll in DLLS:
        project = dll[:-4]
        binary = M / 'artifacts/sdk/bin' / project / 'release' / dll
        shutil.copy2(binary, directory / dll)
        shutil.copy2(binary, OUT / dll)
        binaries.append(dict(path=str(directory / dll), sha256=sha(directory / dll)))
    return dict(name=name, build=receipt, binaries=binaries, source_snapshots=snapshots)

def case(suffix, mode, expected=0, finding=None):
    receipt = operation(suffix, C, ['dotnet', 'bin/Release/net10.0/' + APP, mode, '00:00:12'], expected)
    text = Path(receipt['log']).read_text()
    if mode == 'all':
        assert text.count('\nPASS ') == 41 and 'TOTAL 41 PASS 0 FAIL' in text and '\nFAIL ' not in text
    elif expected == 0:
        assert '\nPASS ' + mode in text and '\nFAIL ' not in text
    else:
        assert '\nFAIL ' + mode + ' Xunit.Sdk.' in text and 'TimeoutException' not in text and 'Case and cleanup failed' not in text
        line = materialization_line if finding == 'F041' else multiset_line
        assert 'RuntimeOwner.cs:line ' + str(line) in text, 'Require finite materialization assertion'
        if finding == 'F041':
            assert 'Xunit.Sdk.NullException' in text and 'same key was already added: ignored-fluent-compensate' in text
        else:
            assert 'Xunit.Sdk.EqualException' in text and 'ENDPOINT_PLAN ' in text
        receipt['finite_materialization_assertion_line'] = line
        receipt['finding'] = finding
    for path, digest in {**oracle, **constants}.items():
        assert sha(path) == digest, path
    return receipt

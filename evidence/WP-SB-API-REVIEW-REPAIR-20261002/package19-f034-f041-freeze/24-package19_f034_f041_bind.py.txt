def join_receipt(record, phase, cwd, command, exit_code):
    actual = [row for row in all_receipts if row['phase'] == phase]
    assert len(actual) == 1
    actual = actual[0]
    assert actual['command'] == command and actual['cwd'] == str(cwd) and actual['exit_code'] == exit_code
    for key, value in actual.items():
        assert record[key] == value
    assert sha(actual['log']) == actual['log_sha256']
    joined_receipts.append(actual)
    return actual

def bind_microsoft(headers, consumer, locked, count, label):
    bound = []
    for name in sorted({row[0] for row in headers if row[0].startswith('Microsoft.Extensions.')}):
        actual = [row for row in headers if row[0] == name]
        assert locked[name]['resolved'] == '10.0.12' and len(actual) == count
        candidates = list((consumer / 'restored-packages' / name.lower() / '10.0.12/lib').glob('*/' + name + '.dll'))
        hashes = {sha(path): str(path) for path in candidates}
        assert all(row[3] in hashes and sha(row[2]) == row[3] for row in actual)
        bound.extend(dict(generation=label, name=row[0], package_version='10.0.12', cache=hashes[row[3]], path=row[2], sha256=row[3]) for row in actual)
    assert bound
    return bound

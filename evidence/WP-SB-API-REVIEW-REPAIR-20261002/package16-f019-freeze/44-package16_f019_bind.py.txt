import ast
import base64
import csv
import hashlib
import json
import re
import zipfile
from pathlib import Path

W = Path('/Users/edgar.liebold/Downloads/ViciOne Suite 2.0')
O = W / 'repositories/vicione-servicebus'
R = Path('/private/tmp/vicione-servicebus-api-review-20261001')
E = O / 'evidence/WP-SB-API-REVIEW-REPAIR-20261002'
D = W / 'SERVICEBUS_API_REVIEW_AND_REPAIR'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
mutpath = E / 'PACKAGE16_F019_MUTATIONS.json'
mut = json.loads(mutpath.read_text())
assert mut['status'] == 'EIGHT_STRICT_COMPILED_SINGLE_EFFECT_MUTANTS_KILLED_WITH_GREEN_CONTROLS'
assert len(mut['mutants']) == 8 and len(mut['controls']) == 2
tree = ast.parse((D / 'package16_f019_mutations.py').read_text())
body = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in ['replace_once', 'mutation']]
paths = mut['frame']
base = {key: (O / rel).read_text() for key, rel in paths.items()}
env = dict(corrected=base)
exec(compile(ast.Module(body=body, type_ignores=[]), '<Root-independent-pure-mutators>', 'exec'), env)
effects = []
for stage in mut['controls'] + mut['mutants']:
    expected = base if stage['name'] in ['corrected', 'rollback'] else env['mutation'](stage['name'])
    assert sha(stage['binary']['path']) == stage['binary']['sha256']
    for item in stage['source_snapshots']:
        digest = hashlib.sha256(expected[item['key']].encode()).hexdigest()
        assert sha(item['path']) == digest == item['sha256'] == item['current_source_sha256']
        effects.append(dict(stage=stage['name'], key=item['key'], snapshot=item['path'], sha256=digest, changed=expected[item['key']] != base[item['key']]))
assert len(effects) == 240
for path, digest in {**mut['oracle_hashes'], **mut['constant_dependencies']}.items():
    assert sha(path) == digest
assert sha(R / 'repair-consumers/package16-f019/corrected/bin/Release/net10.0/ViciOne.ServiceBus.dll') == '42c94cc77fb013c3b3080dc14a1521ab686b7b123ce5d462c6b7e350451d81b7'
rows = list(csv.DictReader((R / 'FILE_COVERAGE.csv').open()))
assert len(rows) == 6209 and all(sha(R / 'mutations/source-main' / row['path']) == row['sha256'] for row in rows)
assert all(not (R / 'mutations/source-main' / rel).exists() for rel in paths.values() if not (R / 'frozen-repository' / rel).exists())
receipts = [json.loads(line) for line in (E / 'VERIFICATION_LOG.jsonl').read_text().splitlines() if json.loads(line)['phase'].startswith('package16-f019-')]
for receipt in receipts:
    assert sha(receipt['log']) == receipt['log_sha256']
    if receipt['exit_code'] == 0 and 'build' in receipt['command']:
        text = Path(receipt['log']).read_text()
        assert '0 Warnung(en)' in text and '0 Fehler' in text
previous = next(x for x in receipts if x['phase'] == 'package16-f019-public-previous-thirtytwo-r2')
corrected = next(x for x in receipts if x['phase'] == 'package16-f019-public-corrected-thirtytwo-r3')
prevtext = Path(previous['log']).read_text()
corrtext = Path(corrected['log']).read_text()
assert previous['exit_code'] == 1 and corrected['exit_code'] == 0
assert prevtext.count('\nFAIL ') == 13 and prevtext.count('\nPASS ') == 19 and 'TOTAL 19 PASS 13 FAIL' in prevtext
assert corrtext.count('\nPASS ') == 32 and 'TOTAL 32 PASS 0 FAIL' in corrtext and '\nFAIL ' not in corrtext
assert not any(s in prevtext + corrtext for s in ['TimeoutException', 'CASE AND OWNED CLEANUP FAILED'])
assert 'FAIL disposeawhilebremainsoperational System.ObjectDisposedException' in prevtext
package_chains = []
microsoft_runtime = []
for mode, logtext in [('previous', prevtext), ('corrected', corrtext)]:
    consumer = R / 'repair-consumers/package16-f019' / mode
    locked = json.loads((consumer / 'packages.lock.json').read_text())['dependencies']['net10.0']
    loaded = {}
    for line in logtext.splitlines():
        if line.startswith('LOADED '):
            name, version, path, digest = line[7:].split(' ', 3)
            assert sha(path) == digest
            loaded[name] = dict(name=name, version=version, path=path, sha256=digest)
    for name in ['ViciOne.ServiceBus', 'ViciOne.ServiceBus.Abstractions']:
        pkg = R / 'repair-packages/package16-f019' / mode / (name + '.1.0.0.nupkg')
        assert base64.b64encode(hashlib.sha512(pkg.read_bytes()).digest()).decode() == locked[name]['contentHash']
        with zipfile.ZipFile(pkg) as z:
            binary = z.read('lib/net10.0/' + name + '.dll')
        digest = hashlib.sha256(binary).hexdigest()
        cached = consumer / 'restored-packages' / name.lower() / '1.0.0/lib/net10.0' / (name + '.dll')
        assert sha(cached) == digest == loaded[name]['sha256']
        package_chains.append(dict(consumer=mode, package=str(pkg), package_sha256=sha(pkg), locked_sha512=locked[name]['contentHash'], cache=str(cached), loaded=loaded[name]))
    for name, item in loaded.items():
        if name.startswith('Microsoft.Extensions.'):
            assert locked[name]['resolved'] == '10.0.12'
            folder = consumer / 'restored-packages' / name.lower() / '10.0.12'
            candidates = [p for p in (folder / 'lib').glob('*/' + name + '.dll') if sha(p) == item['sha256']]
            assert candidates, name
            microsoft_runtime.append(dict(consumer=mode, package_version='10.0.12', cache=str(candidates[0]), **item))
binding = dict(status='ROOT_INDEPENDENT_EFFECTS_RECEIPTS_PACKAGE_RUNTIME_RESTORATION_VERIFIED', source_effect_checks=effects, actual_receipts=receipts, selected_package_chains=package_chains, actual_Microsoft_runtime=microsoft_runtime, original_cases=dict(causal_reds=13, assertion_reds=12, causal_business_ODE=1, positive=19, guards=0), corrected_cases=32, all6209_original_files_restored=True, oracle_and_runtime_restored=True, limits='PE/PDB source bindings independently required in final adversarial review. No exactSDKsource FULL/canonical766/heap/ARM/cloud/wholeAPI claim.')
bp = E / 'PACKAGE16_F019_ROOT_BINDING.json'
assert not bp.exists()

directory = E / 'package16-f019-freeze'
directory.mkdir(exist_ok=False)
owned = []
inputs = [O / rel for rel in paths.values()]
inputs += [p for mode in ['previous', 'corrected'] for p in sorted((R / 'repair-consumers/package16-f019' / mode).iterdir()) if p.is_file()]

    owned.append(dict(path=str(path), snapshot=str(snapshot), bytes=path.stat().st_size, sha256=sha(path)))
plans = [dict(path=str(p), sha256=sha(p)) for p in sorted(E.glob('PACKAGE16_F019_*.json')) if p not in [bp, mutpath]]
freeze = dict(status='FROZEN_SCOPED_F019_FINAL_ADVERSARIAL_PENDING', finding='F-SB-API-019', owned_inputs=owned, root_binding=dict(path=str(bp), sha256=sha(bp)), mutations=dict(path=str(mutpath), sha256=sha(mutpath)), plans_and_exclusions=plans, limits='32 fresh-process public cases; one normal in-memory bus per default/typed route, two buses explicitly in both route. Logging/DI Nuget10.0.12 actual copied/runtime hashes; Framework10.0.10 CoreLib. No universal quiescence after guardfailure/heap/ARM/cloud/wholecanonical/API/A+ grade.')

assert not fp.exists()

print(json.dumps(dict(freeze=str(fp), sha256=sha(fp), owned=len(owned), source_effects=len(effects), receipts=len(receipts), selected_package_chains=len(package_chains), Microsoft_runtime_bindings=len(microsoft_runtime))))

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Diagnostics.Metrics;
using System.Threading;
using Microsoft.Extensions.DependencyInjection;
using ViciOne.ServiceBus.Logging.Diagnostics;
using ViciOne.ServiceBus.Metadata;
using ViciOne.ServiceBus.Monitoring;

namespace ViciOne.ServiceBus.MessageJournal;

internal sealed class MessageJournalTelemetry : IDisposable
{
    internal readonly record struct Scope(Activity? Activity, Activity? Parent);

    private const string FailedResult = "failed";
    private const string FilteredResult = "filtered";
    private const string StoredResult = "stored";

    private static readonly Lazy<MessageJournalTelemetry> ProcessTelemetry = new(
        static () => new MessageJournalTelemetry(null, processOwned: true),
        LazyThreadSafetyMode.ExecutionAndPublication);

    private readonly ActivitySource? _activitySource;
    private readonly Counter<long>? _operations;
    private readonly Histogram<double>? _duration;
    private readonly Meter? _processMeter;
    private int _disposed;

    internal static MessageJournalTelemetry Process => ProcessTelemetry.Value;

    internal static MessageJournalTelemetry Create(IServiceProvider provider)
    {
        try
        {
            return new MessageJournalTelemetry(provider.GetRequiredService<IMeterFactory>(), processOwned: false);
        }
        catch
        {
            // A failed DI activation must never select process-global instrumentation.
            return new MessageJournalTelemetry(null, processOwned: false);
        }
    }

    private MessageJournalTelemetry(IMeterFactory? meterFactory, bool processOwned)
    {
        if (meterFactory is null && !processOwned)
            return;

        ActivitySource? source = null;
        Meter? processMeter = null;
        try
        {
            string? version = HostMetadataCache.Host.ViciOneServiceBusVersion;
            source = new ActivitySource(ServiceBusTelemetry.ActivitySourceName, version);
            Meter meter = processOwned
                ? processMeter = new Meter(ServiceBusTelemetry.MeterName, version)
                : meterFactory!.Create(new MeterOptions(ServiceBusTelemetry.MeterName) { Version = version });
            _operations = meter.CreateCounter<long>(
                ServiceBusTelemetry.Metrics.MessageJournalOperations,
                "{operation}",
                "Completed message-journal observations.");
            _duration = meter.CreateHistogram<double>(
                ServiceBusTelemetry.Metrics.MessageJournalDuration,
                "s",
                "Duration of processing one message-journal observation.");
            _activitySource = source;
            _processMeter = processMeter;
        }
        catch
        {
            _operations = null;
            _duration = null;
            try { source?.Dispose(); } catch { }
            try { processMeter?.Dispose(); } catch { }
        }
    }

    public Scope StartActivity(
        MessageJournalOperation operation,
        MessageJournalOutcome outcome)
    {
        Activity? parent = Activity.Current;
        if (Volatile.Read(ref _disposed) != 0 || _activitySource is null)
            return new Scope(null, parent);
        try
        {
            TagList tags = CreateTags(operation, outcome, result: null);
            System.Diagnostics.ActivityContext parentContext = parent?.Context ?? default;
            Activity? activity = ActivityObservation.TryStartSource(_activitySource,
                ServiceBusTelemetry.Activities.MessageJournalObserve,
                ActivityKind.Internal,
                parentContext,
                tags);
            return new Scope(activity, parent);
        }
        catch (Exception)
        {
            return new Scope(null, parent);
        }
    }

    public void Stored(
        Scope scope,
        MessageJournalOperation operation,
        MessageJournalOutcome outcome,
        TimeSpan duration)
    {
        Complete(scope, operation, outcome, StoredResult, failureReason: null, duration);
    }

    public void Filtered(
        Scope scope,
        MessageJournalOperation operation,
        MessageJournalOutcome outcome,
        TimeSpan duration)
    {
        Complete(scope, operation, outcome, FilteredResult, failureReason: null, duration);
    }

    public void Failed(
        Scope scope,
        MessageJournalOperation operation,
        MessageJournalOutcome outcome,
        string reason,
        TimeSpan duration)
    {
        Complete(scope, operation, outcome, FailedResult, reason, duration);
    }

    private void Complete(
        Scope scope,
        MessageJournalOperation operation,
        MessageJournalOutcome outcome,
        string result,
        string? failureReason,
        TimeSpan duration)
    {
        Activity? activity = scope.Activity;
        try
        {
            TagList tags = CreateTags(operation, outcome, result);
            if (failureReason is not null)
                tags.Add(ServiceBusTelemetry.Attributes.MessageJournalFailureReason, failureReason);

            ApplyTags(activity, tags);
            activity?.SetStatus(result == FailedResult ? ActivityStatusCode.Error : ActivityStatusCode.Ok, failureReason);
            if (Volatile.Read(ref _disposed) == 0)
            {
                _operations?.Add(1, tags);
                _duration?.Record(duration.TotalSeconds, tags);
            }
        }
        catch (Exception)
        {
            // Telemetry observers never own message or journal semantics.
        }
        finally
        {
            if (activity is not null)
                ActivityObservation.TryDispose(activity, scope.Parent);
        }
    }

    private static TagList CreateTags(
        MessageJournalOperation operation,
        MessageJournalOutcome outcome,
        string? result)
    {
        TagList tags = default;
        tags.Add(ServiceBusTelemetry.Attributes.MessageJournalOperation, operation.ToString().ToLowerInvariant());
        tags.Add(ServiceBusTelemetry.Attributes.MessageJournalOutcome, outcome.ToString().ToLowerInvariant());
        if (result is not null)
            tags.Add(ServiceBusTelemetry.Attributes.MessageJournalResult, result);

        return tags;
    }

    private static void ApplyTags(Activity? activity, TagList tags)
    {
        if (activity is null)
            return;

        foreach (KeyValuePair<string, object?> tag in tags)
            activity.SetTag(tag.Key, tag.Value);
    }

    public void Dispose()
    {
        if (Interlocked.Exchange(ref _disposed, 1) != 0)
            return;

        try { _activitySource?.Dispose(); } catch { }
        try { _processMeter?.Dispose(); } catch { }
    }
}

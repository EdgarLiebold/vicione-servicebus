namespace ViciOne.ServiceBus.RabbitMqTransport.Tests
{
    using System;
    using System.Collections.Generic;
    using System.Diagnostics;
    using System.Threading.Tasks;
    using NUnit.Framework;
    using RabbitMQ.Client.Exceptions;
    using ViciOne.ServiceBus.Testing;


    /// <summary>
    /// Reproduces what the broker and the transport actually do when a bus endpoint queue is already
    /// held exclusively, so the decision on F-3 rests on a measurement instead of on a reading of the
    /// code. It asserts nothing about the desired behaviour — it records the present one.
    /// </summary>
    [TestFixture]
    [Explicit("Diagnostic probe for record 0035 F-3. Records the current behaviour; it is not a gate.")]
    public class Probing_the_exclusive_conflict
    {
        [Test]
        public async Task Should_report_what_the_second_bus_actually_receives()
        {
            var harness1 = new RabbitMqTestHarness();
            harness1.OnConfigureRabbitMqBus += configurator =>
            {
                configurator.OverrideDefaultBusEndpointQueueName("exclusively-yours");
                configurator.Exclusive = true;
            };

            var harness2 = new RabbitMqTestHarness();
            harness2.OnConfigureRabbitMqBus += configurator =>
            {
                configurator.OverrideDefaultBusEndpointQueueName("exclusively-yours");
                configurator.Exclusive = true;
            };

            await harness1.RecreateVirtualHost();
            await harness1.Start();

            harness1.SubscribeHandler<TestFramework.Messages.PingMessage>();
            await Task.Delay(2000);

            try
            {
                await harness2.Start();

                var stopwatch = Stopwatch.StartNew();
                Exception captured = null;

                try
                {
                    harness2.SubscribeHandler<TestFramework.Messages.PingMessage>();
                }
                catch (Exception exception)
                {
                    captured = exception;
                }

                stopwatch.Stop();

                TestContext.Out.WriteLine($"elapsed: {stopwatch.Elapsed}");
                TestContext.Out.WriteLine($"outcome: {(captured == null ? "no exception" : captured.GetType().FullName)}");

                foreach (var exception in Chain(captured))
                {
                    TestContext.Out.WriteLine($"  {exception.GetType().FullName}: {exception.Message}");

                    if (exception is OperationInterruptedException interrupted)
                    {
                        TestContext.Out.WriteLine($"    replyCode={interrupted.ShutdownReason?.ReplyCode} "
                            + $"replyText={interrupted.ShutdownReason?.ReplyText} "
                            + $"classId={interrupted.ShutdownReason?.ClassId} methodId={interrupted.ShutdownReason?.MethodId}");
                    }
                }

                await harness2.Stop();
            }
            finally
            {
                await Task.Delay(1000);

                await harness1.Stop();
            }
        }

        static IEnumerable<Exception> Chain(Exception exception)
        {
            while (exception != null)
            {
                yield return exception;
                exception = exception.InnerException;
            }
        }
    }
}
